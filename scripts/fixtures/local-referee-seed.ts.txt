export const alter = internalMutation({
  args: { matchId: v.id("matches"), mode: v.string(), periods: v.optional(v.number()), minutes: v.optional(v.number()), margin: v.optional(v.number()), generation: v.optional(v.number()) },
  handler: async (ctx, args) => {
    const match = await ctx.db.get(args.matchId);
    if (!match?.publicCode.startsWith("LOCAL_ONLY_") || !match.refereePolicy) throw new Error("LOCAL_FIXTURE_REQUIRED");
    const referee = await ctx.db.query("referees").withIndex("by_email", q => q.eq("email", "ref@localhost.test")).first();
    const policy = match.refereePolicy; const now = Date.now();
    if (args.mode === "unassign") await ctx.db.patch(match._id, { refereeId: undefined });
    else if (args.mode === "assign") await ctx.db.patch(match._id, { refereeId: referee!._id });
    else if (args.mode === "cap") {
      const periods = args.periods ?? 2, minutes = args.minutes ?? 30, margin = args.margin ?? 10;
      const rows = await ctx.db.query("matchPlayers").withIndex("by_match", q => q.eq("matchId", match._id)).collect();
      for (const row of rows) await ctx.db.patch(row._id, { minutesPlayed: 0, officialEnteredElapsedMs: row.onField ? 0 : undefined,
        lastSubbedInAt: row.onField ? now - 600 * 60000 : undefined });
      await ctx.db.patch(match._id, { status: "live", currentQuarter: 1, quarterCount: periods, regulationDurationMinutes: periods * minutes, startedAt: now - 600 * 60000,
        refereePolicy: { ...policy, generation: policy.generation + 1, elapsedMs: 0, runningSince: now - 600 * 60000, periodStartedElapsedMs: 0, safetyMarginMinutes: margin, confirmedStoppageMs: 0, clockReviewReason: undefined, stoppageStartedAt: undefined, clockLimitAt: undefined } });
    } else if (args.mode === "healthy") {
      await ctx.db.patch(match._id, { status: "live", currentQuarter: 1, quarterCount: 2, regulationDurationMinutes: 60,
        refereePolicy: { ...policy, generation: policy.generation + 1, elapsedMs: 0, runningSince: now, periodStartedElapsedMs: 0, safetyMarginMinutes: 10, confirmedStoppageMs: 0, clockReviewReason: undefined, stoppageStartedAt: undefined, clockLimitAt: undefined } });
    } else if (args.mode === "long_pause") {
      await ctx.db.patch(match._id, { status: "live", currentQuarter: 1, quarterCount: 2, regulationDurationMinutes: 60,
        refereePolicy: { ...policy, generation: policy.generation + 1, elapsedMs: 12 * 60000, runningSince: null, periodStartedElapsedMs: 0,
          safetyMarginMinutes: 10, confirmedStoppageMs: 0, clockReviewReason: undefined, stoppageStartedAt: now - 600 * 60000, clockLimitAt: undefined } });
    } else if (args.mode === "revoke" || args.mode === "restore") {
      const scope = await ctx.db.query("mobilePilotScopes").withIndex("by_subject_and_match", q => q.eq("subject", "local-referee").eq("matchId", match._id)).unique();
      await ctx.db.patch(scope!._id, { enabled: args.mode === "restore" });
    } else throw new Error("UNKNOWN_FIXTURE_OPERATION");
    return null;
  },
});
